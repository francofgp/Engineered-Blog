{{- /*
  llms.txt — emerging standard (https://llmstxt.org) for telling LLMs which content on the
  site is worth indexing. Wired up by:
    - config.toml: [outputFormats.LLMSTXT] (defines mediaType, baseName, isPlainText)
    - config.toml: [outputs].home includes "LLMSTXT" (only emitted on the home page)
    - This file: rendered to /llms.txt at site root.
  Spec format (strict order):
    1. H1 with site name (required)
    2. Blockquote with short summary
    3. Optional paragraphs of context
    4. H2 sections, each containing markdown list of [name](url): description entries
  An "## Optional" H2 has special meaning — LLMs may skip those URLs when context is tight.
*/ -}}
{{- /* Build a comma-separated list of the top 5 categories by post count. .ByCount returns
       an OrderedTaxonomy ([]OrderedTaxonomyEntry); we range and accumulate .Name into a
       slice, then delimit. Falls back to a sensible static list if no categories exist.
       Computed at top so the body below can use it without {{-/-}} eating the newlines
       between markdown blocks (blockquote → paragraph → H2) — that was the bug fixed in
       the audit: previously the {{- $topCats := -}} block sat between the blockquote and
       the next paragraph and swallowed the blank line, collapsing 528 chars into one line. */ -}}
{{- $topCats := slice -}}
{{- range first 5 site.Taxonomies.categories.ByCount -}}
  {{- $topCats = $topCats | append .Name -}}
{{- end -}}
